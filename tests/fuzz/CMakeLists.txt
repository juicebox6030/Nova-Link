# libFuzzer harnesses (NOVA_FUZZ=ON, clang) or, in normal test builds, plain
# replay binaries that run the committed seeds and crash regressions as tests.
set(nova_fuzzers transport radio)
if(NOVA_SECURITY)
    list(APPEND nova_fuzzers secure)
endif()

if(NOVA_FUZZ)
    if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "NOVA_FUZZ needs clang (libFuzzer)")
    endif()
    set(nova_fuzz_flags -fsanitize=fuzzer-no-link,address,undefined
        -fno-sanitize-recover=undefined -fno-omit-frame-pointer -g)
    target_compile_options(nova_link PRIVATE ${nova_fuzz_flags})
    foreach(name ${nova_fuzzers})
        add_executable(fuzz_${name} fuzz_${name}.c)
        target_link_libraries(fuzz_${name} PRIVATE nova_link)
        nova_target(fuzz_${name})
        target_compile_options(fuzz_${name} PRIVATE ${nova_fuzz_flags})
        target_link_options(fuzz_${name} PRIVATE -fsanitize=fuzzer,address,undefined)
    endforeach()
else()
    foreach(name ${nova_fuzzers})
        add_executable(replay_${name} fuzz_${name}.c standalone/main.c)
        target_link_libraries(replay_${name} PRIVATE nova_link)
        nova_target(replay_${name})
        file(GLOB seeds ${CMAKE_CURRENT_SOURCE_DIR}/seeds/${name}/*)
        if(seeds)
            add_test(NAME fuzz_seeds_${name} COMMAND replay_${name} ${seeds})
        endif()
    endforeach()
endif()
